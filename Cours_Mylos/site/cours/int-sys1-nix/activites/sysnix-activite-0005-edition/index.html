1.2

Rechercher dans l’aide de vim la commande pour faire un “rechercher/remplacer”.

2

Écrire le programme beuseu.c ci-dessous et compiler le

/*
 ***********************
 dominique huguenin (dominique.huguenin@rpn.ch)
 ************************
*/

#include <stdio.h>
#include <stdlib.h> 

#define LIMITE_MAX 20
#define NUMERO_INTERDIT 7
#define MOT_MAGIQUE "Beuseu"

int main(int argc, char* argv[]){
  int limite;
  int nombreTemporaire;
  int unite;
  int multiple7;
  int contient7;
  int nombre;

  if ( argc <= 1 ){
        limite = LIMITE_MAX;
  } else {
        limite = atoi(argv[1]);
  }

  for ( nombre = 1; nombre <= limite ; nombre++ ){
    multiple7 = nombre % NUMERO_INTERDIT == 0;
    nombreTemporaire = nombre;

    do {
      unite = nombreTemporaire % 10;
      contient7 = ( unite % NUMERO_INTERDIT ) == 0 && ( unite != 0 );
      nombreTemporaire = nombreTemporaire / 10;
    } while ( !contient7 && (nombreTemporaire > 0 ) ); 

    if ( multiple7 || contient7 ) {
      printf("%s\n",MOT_MAGIQUE);
    } else {
      printf("%d\n",nombre);
    }
  }
  return 0;
}

3

Sauvegarder le fichier beuseu.c dans le dossier unix de votre dossier personnel sur S2.

4

Compiler avec le compilateur gcc le fichier beuseu.c. La compilation doit produire un exécutable du nom de beuseu. Le code beuseu.c est du C ANSI.

5

Configurer le système pour que la commande ci-dessous fonctionne:

$ beuseu
